const DB_NAME = 'LexForensicaVault';
const STORE_NAME = 'encryptedCases';

class SecureStorage {
  static async initDB() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(DB_NAME, 1);
      request.onupgradeneeded = (e) => {
        const db = e.target.result;
        if (!db.objectStoreNames.contains(STORE_NAME)) {
          db.createObjectStore(STORE_NAME, { keyPath: 'id' });
        }
      };
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  static async deriveKey(password, salt) {
    const enc = new TextEncoder();
    const keyMaterial = await window.crypto.subtle.importKey(
      'raw', enc.encode(password), { name: 'PBKDF2' }, false, ['deriveBits', 'deriveKey']
    );
    return window.crypto.subtle.deriveKey(
      { name: 'PBKDF2', salt: salt, iterations: 100000, hash: 'SHA-256' },
      keyMaterial, { name: 'AES-GCM', length: 256 }, true, ['encrypt', 'decrypt']
    );
  }

  static async encryptData(data, password) {
    const salt = window.crypto.getRandomValues(new Uint8Array(16));
    const iv = window.crypto.getRandomValues(new Uint8Array(12));
    const key = await this.deriveKey(password, salt);
    const enc = new TextEncoder();
    const cipher = await window.crypto.subtle.encrypt(
      { name: 'AES-GCM', iv: iv }, key, enc.encode(JSON.stringify(data))
    );
    return { cipher, iv, salt };
  }

  static async decryptData(encryptedObj, password) {
    try {
      const key = await this.deriveKey(password, encryptedObj.salt);
      const plainBuffer = await window.crypto.subtle.decrypt(
        { name: 'AES-GCM', iv: encryptedObj.iv }, key, encryptedObj.cipher
      );
      const dec = new TextDecoder();
      return JSON.parse(dec.decode(plainBuffer));
    } catch (e) {
      throw new Error('Neplatné dešifrovací heslo.');
    }
  }
}

// 2. FILE PROCESSOR (Lokální extrakce textu)
// ==========================================
class FileProcessor {
  static async extractText(file) {
    if (file.type === 'application/pdf') {
      return await this.extractPdfText(file);
    } else {
      // Pro txt, csv, atd.
      return await file.text();
    }
  }

  static async extractPdfText(file) {
    // Dynamické načtení pdf.js pro lokální zpracování bez nutnosti bundlingu
    if (!window.pdfjsLib) {
      await new Promise((resolve, reject) => {
        const script = document.createElement('script');
        script.src = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.min.js';
        script.onload = () => {
          window.pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js';
          resolve();
        };
        script.onerror = () => reject(new Error("Nepodařilo se načíst PDF.js parser."));
        document.head.appendChild(script);
      });
    }

    try {
      const arrayBuffer = await file.arrayBuffer();
      const pdf = await window.pdfjsLib.getDocument(arrayBuffer).promise;
      let fullText = '';
      for (let i = 1; i <= pdf.numPages; i++) {
        const page = await pdf.getPage(i);
        const textContent = await page.getTextContent();
        fullText += textContent.items.map(s => s.str).join(' ') + '\n';
      }
      return fullText;
    } catch (e) {
      console.error("PDF Parsing error:", e);
      throw new Error(`Nepodařilo se přečíst PDF: ${file.name}`);
    }
  }
}

const apiKey = ""; // API klíč je automaticky dodán exekučním prostředím

class GeminiService {
  static async analyzeDocuments(filesData, onLog) {
    // 1. Spojení textů s identifikátory zdrojů
    const combinedText = filesData.map(f => `--- ZDROJ: ${f.name} ---\n${f.text}`).join('\n\n');

    onLog("Kompletuji textový korpus a odesílám LLM instrukce...");

    const systemPrompt = `Jsi LEX FORENSICA, expertní AI auditor pro medicínsko-právní případy. 
Tento systém hodnotí případy v kontextu českého práva (zák. 372/2011 Sb.), Evropské charty a Úmluvy OSN o právech osob se zdravotním postižením (CRPD).
Tvým úkolem je analyzovat poskytnuté zdravotnické texty, extrahovat chronologii případu, detekovat kognitivní zkreslení (bias), identifikovat právní pochybení a vygenerovat strukturovaný výstup.
Vždy vrať pouze JSON odpovídající definovanému schématu. Zjištění musí být faktická a podložená extrahovaným textem.`;

    const schema = {
      type: "OBJECT",
      properties: {
        caseId: { type: "STRING", description: "Vygenerované ID případu, např. CASE-2025-XYZ" },
        metadata: {
          type: "OBJECT",
          properties: {
            patientCode: { type: "STRING" },
            jurisdiction: { type: "STRING" },
            status: { type: "STRING" }
          }
        },
        chronology: {
          type: "ARRAY",
          items: {
            type: "OBJECT",
            properties: {
              id: { type: "STRING" },
              date: { type: "STRING", description: "Datum ve formátu ISO" },
              type: { type: "STRING", description: "admission, intervention, observation, legal" },
              title: { type: "STRING" },
              description: { type: "STRING" },
              source: { type: "STRING", description: "Název původního souboru" },
              confidence: { type: "NUMBER", description: "Od 0.0 do 1.0" },
              legalFlags: { type: "ARRAY", items: { type: "STRING" } },
              biasFlags: { type: "ARRAY", items: { type: "STRING" } },
              entities: { type: "ARRAY", items: { type: "STRING" } }
            }
          }
        },
        legalMatrix: {
          type: "ARRAY",
          items: {
            type: "OBJECT",
            properties: {
              id: { type: "STRING" },
              law: { type: "STRING" },
              description: { type: "STRING" },
              violationRisk: { type: "STRING", description: "HIGH, CRITICAL, nebo LOW" },
              evidenceRef: { type: "STRING", description: "id události z chronologie" },
              rationale: { type: "STRING" }
            }
          }
        },
        humanReviewQueue: {
          type: "ARRAY",
          items: {
            type: "OBJECT",
            properties: {
              id: { type: "STRING" },
              relatedEventId: { type: "STRING" },
              question: { type: "STRING", description: "Otázka pro lidského experta k validaci nejistoty nebo sporu" },
              status: { type: "STRING", description: "Vždy PENDING" },
              aiConfidence: { type: "NUMBER" }
            }
          }
        }
      },
      required: ["caseId", "metadata", "chronology", "legalMatrix", "humanReviewQueue"]
    };

    const payload = {
      contents: [{ parts: [{ text: combinedText }] }],
      systemInstruction: { parts: [{ text: systemPrompt }] },
      generationConfig: {
        responseMimeType: "application/json",
        responseSchema: schema
      }
    };

    // Exponenciální backoff
    let retries = 0;
    const maxRetries = 5;
    const delays = [1000, 2000, 4000, 8000, 16000];

    onLog("Fáze ANALYZE & EVALUATE: Spouštím AI Inferenci (Gemini)...");

    while (retries <= maxRetries) {
      try {
        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        
        if (!response.ok) {
           const errText = await response.text();
           throw new Error(`API Error ${response.status}: ${errText}`);
        }
        
        const data = await response.json();
        const textResult = data.candidates[0].content.parts[0].text;
        
        onLog("AI analýza úspěšná. Formátuji a strukturuji forenzní výstupy.");
        return JSON.parse(textResult);
        
      } catch (err) {
        onLog(`Chyba API. Pokus o retry č. ${retries + 1}...`);
        if (retries === maxRetries) throw err;
        await new Promise(r => setTimeout(r, delays[retries]));
        retries++;
      }
    }
  }
}

// 1. UPLOAD VIEW (Ostré zpracování souborů a volání LLM)
// ------------------------------------------
function UploadView({ onComplete }) {
  const [files, setFiles] = useState([]);
  const [isProcessing, setIsProcessing] = useState(false);
  const [logs, setLogs] = useState([]);
  const fileInputRef = useRef(null);

  const handleFileChange = (e) => {
    const selectedFiles = Array.from(e.target.files);
    setFiles(prev => [...prev, ...selectedFiles]);
  };

  const removeFile = (index) => {
    setFiles(prev => prev.filter((_, i) => i !== index));
  };

  const addLog = (msg) => {
    setLogs(prev => [...prev, `[${new Date().toLocaleTimeString()}] ${msg}`]);
  };

  const startAnalysis = async () => {
    if (files.length === 0) return alert("Prosím nahrajte alespoň jeden důkazní materiál.");
    
    setIsProcessing(true);
    setLogs([]);
    
    try {
      addLog("Začínám bezpečné lokální čtení souborů...");
      const extractedFiles = [];
      
      for (const file of files) {
        addLog(`Čtu soubor: ${file.name}...`);
        const text = await FileProcessor.extractText(file);
        extractedFiles.push({ name: file.name, text });
      }

      const generatedData = await GeminiService.analyzeDocuments(extractedFiles, addLog);
      
      setIsProcessing(false);
      onComplete(generatedData);

    } catch (error) {
      console.error(error);
      addLog(`KRITICKÁ CHYBA: ${error.message}`);
      setIsProcessing(false);
    }
  };

  return (
    <div className="max-w-4xl mx-auto space-y-6">
      <div className="bg-white p-8 rounded-xl shadow-sm border border-slate-200">
        <h2 className="text-2xl font-bold text-slate-800 mb-2">Nahrání Důkazních Materiálů</h2>
        <p className="text-slate-500 mb-6">Nahrajte lékařské zprávy, posudky, ošetřovatelskou dokumentaci (PDF, TXT, CSV). Systém z nich extrahuje časovou osu a identifikuje právní pochybení pomocí LLM.</p>
        
        {!isProcessing && (
          <div 
            onClick={() => fileInputRef.current.click()}
            className="border-2 border-dashed border-blue-300 bg-blue-50 hover:bg-blue-100 transition-colors rounded-xl p-10 flex flex-col items-center justify-center cursor-pointer mb-6 text-center"
          >
            <input type="file" multiple className="hidden" ref={fileInputRef} onChange={handleFileChange} accept=".pdf,.txt,.csv" />
            <div className="text-blue-500 mb-3"><Icons.UploadCloud /></div>
            <div className="font-semibold text-blue-700 mb-1">Klikněte nebo přetáhněte soubory sem</div>
            <div className="text-sm text-blue-500">Podporováno: PDF, TXT, CSV (zpracováno lokálně)</div>
          </div>
        )}

        {files.length > 0 && !isProcessing && (
          <div className="mb-6">
            <h3 className="text-sm font-bold text-slate-700 mb-3">Fronta ke zpracování ({files.length} souborů)</h3>
            <ul className="space-y-2">
              {files.map((file, i) => (
                <li key={i} className="flex items-center justify-between bg-slate-50 p-3 rounded-lg border border-slate-200">
                  <div className="flex items-center gap-3">
                    <span className="text-slate-400"><Icons.FileText /></span>
                    <span className="text-sm font-medium text-slate-700">{file.name}</span>
                    <span className="text-xs text-slate-400">{(file.size / 1024).toFixed(1)} KB</span>
                  </div>
                  <button onClick={() => removeFile(i)} className="text-slate-400 hover:text-red-500 transition-colors">
                    <Icons.Trash />
                  </button>
                </li>
              ))}
            </ul>
            
            <div className="mt-6 flex justify-end">
              <button onClick={startAnalysis} className="px-6 py-3 bg-slate-900 hover:bg-slate-800 text-white font-medium rounded-lg shadow-md transition-colors flex items-center gap-2">
                <Icons.Brain /> Odeslat do AI Forenzní Sítě
              </button>
            </div>
          </div>
        )}

        {isProcessing && (
          <div className="bg-slate-950 rounded-xl p-6 overflow-hidden border border-slate-800 shadow-inner">
            <div className="flex items-center gap-3 mb-4 border-b border-slate-800 pb-4">
              <span className="text-blue-500"><Icons.Loader /></span>
              <h3 className="text-blue-400 font-mono font-bold text-sm">LEX FORENSICA ENGINE BĚŽÍ...</h3>
            </div>
            <div className="space-y-2 font-mono text-xs text-green-400 min-h-[150px]">
              {logs.map((log, i) => (
                <div key={i} className={i === logs.length - 1 ? "animate-pulse" : ""}>{log}</div>

        )}

    </div>
  );
}

function BiasView({ events }) {
  if (!events) return null;
  const eventsWithBias = events.filter(e => e.biasFlags && e.biasFlags.length > 0);
  return (
    <div className="max-w-5xl mx-auto py-4">
      <h2 className="text-2xl font-bold text-slate-800 mb-6">Sémantický Posun a Zkreslení</h2>
      {eventsWithBias.length === 0 ? <p className="text-slate-500">Žádné zkreslení nebylo detekováno.</p> : null}
      {eventsWithBias.map(event => (
        <div key={event.id} className="bg-white rounded-xl border border-purple-200 shadow-sm p-6 mb-4 border-l-4 border-l-purple-500">
          <div className="font-bold text-slate-800">{event.title}</div>
          <div className="text-sm text-slate-600 mt-2">{event.description}</div>
          <div className="mt-3 flex gap-2">
            {event.biasFlags.map(f => <span key={f} className="bg-purple-100 text-purple-800 text-xs px-2 py-1 rounded">{f}</span>)}
          </div>
        </div>
      ))}
    </div>
  );
}

function ReviewView({ queue, onResolve }) {
  if (!queue || queue.length === 0) return <div className="p-6">Fronta revizí je prázdná.</div>;
  return (
    <div className="max-w-4xl mx-auto py-4">
      <h2 className="text-2xl font-bold text-slate-800 mb-6">Lidská Revize (Human-In-The-Loop)</h2>
      {queue.map(item => (
        <div key={item.id} className={`p-6 bg-white border rounded-xl shadow-sm ${item.status === 'RESOLVED' ? 'border-green-200 opacity-50' : 'border-blue-200'}`}>
          <div className="font-medium text-lg mb-4">{item.question}</div>
          {item.status !== 'RESOLVED' && (
            <div className="flex gap-3">
              <button onClick={() => onResolve(item.id)} className="px-4 py-2 bg-green-600 text-white rounded text-sm">Potvrdit</button>
              <button onClick={() => onResolve(item.id)} className="px-4 py-2 bg-red-100 text-red-700 rounded text-sm">Zamítnout</button>
            </div>
          )}
        </div>
      ))}
    </div>
  );
}
